Pass work to functions, choose results with patterns, and repeat work with recursion.

```i run package=tour title="src/main.i"
(
	i/core
	i/stdio
)

square = (x) { x * x }
main = () {
	stdio.print(
		core.inspect(square(3)),
	)
}
```

```text
9
```

Parameter and result types can be inferred.

## Function arguments

```i run package=tour title="src/main.i"
(
	i/core
	i/stdio
)

apply = (f, x) { f(x) }
square = (x) { x * x }

main = () {
	result = apply(square, 3)
	stdio.print(core.inspect(result))
}
```

```text
9
```

A function name supplies the function value; a call supplies its result.

## Labelled arguments

```i run package=tour title="src/main.i"
(
	i/stdio
)

greet = (name, punctuation = ending) { "Hello, \(name)\(ending)" }

main = () { stdio.print(greet("Ada", punctuation = "!")) }
```

```text
Hello, Ada!
```

`punctuation` is the caller’s label; `ending` is the local name, with no default value.

## Blocks

```i run package=tour title="src/main.i"
(
	i/core
	i/stdio
)

main = () {
	result = {
		first = 3 + 2
		first * 2
	}
	stdio.print(core.inspect(result))
}
```

```text
10
```

The final statement must be an expression; braces also group expressions, while parentheses make data.

## Branches

```i run package=tour title="src/main.i"
(
	i/core
	i/stdio
)

size = (n : Int) {
	n |> (0 | 1) { "small" } | (x ? x < 0) { "negative" } | (_) { "large" }
}

main = () {
	stdio.print(core.inspect((size(0), size(1), size(-2), size(5))))
}
```

```text
("small", "small", "negative", "large")
```

A chain tries branches in order; `?` tests a guard after matching.

Guards do not prove coverage; `_` covers the remaining inputs, and alternative patterns must bind the same names.

## Type tests

```i run package=tour title="src/main.i"
(
	i/core
	i/stdio
)

kind = (value : { Int | String }) {
	value |> (n : Int) { core.inspect(n) } | (s : String) { s }
}

main = () {
	stdio.print(core.inspect((kind(7), kind("text"))))
}
```

```text
("7", "text")
```

In a chain, the annotations test union members; a single ordinary parameter annotation instead constrains the accepted argument.

## Conditions

```i run package=tour title="src/main.i"
(
	i/stdio
)

main = () {
	{ 2 < 3 } |> (True) { stdio.print("yes") } | (False) { stdio.print("no") }
}
```

```text
yes
```

Choices need no control-flow keywords.

## Tuple patterns

```i run package=tour title="src/main.i"
(
	i/core
	i/stdio
)

main = () {
	pair = (4, 5)
	(a, __unused) = pair
	(left, right) = whole = pair
	stdio.print(core.inspect((a, left, right, whole)))
}
```

```text
(4, 4, 5, (4, 5))
```

`__unused` discards without binding; the conjunction matches one value with both patterns.

## Constructor patterns

```i run package=tour title="src/main.i"
(
	i/core
	i/stdio
)

Item = (value = Int)
Empty = ()
unpack = (value : { Item | Empty }) {
	value |> (Item(value = n)) { n } | (Empty) { 0 }
}

main = () {
	stdio.print(core.inspect((unpack(Item(value = 8)), unpack(Empty))))
}
```

```text
(8, 0)
```

Labelled constructor fields need labels in their patterns.

## List patterns

```i run package=tour title="src/main.i"
(
	i/core
	i/stdio
)

list_parts = (xs : List(Int)) {
	xs |> ([]) { (0, []) } | ([x]) { (x, []) } | ([x, ..rest]) { (x, rest) }
}

main = () {
	stdio.print(core.inspect((list_parts([]), list_parts([2]), list_parts([2, 3]))))
}
```

```text
((0, []), (2, []), (2, [3]))
```

Current limits: keyed data patterns (DOC1) and data-rest patterns (P13) are rejected; use field access and list rests ([known gaps](https://github.com/nmnmcc/i/blob/main/docs/design/spec-known-gaps.md)).

## Text patterns

```i run package=tour title="src/main.i"
(
	i/core
	i/stdio
)

prefix = (s : String) { s |> ("pre:" <> rest) { rest } | (_) { "no prefix" } }
suffix = (s : String) { s |> (rest <> ".i") { rest } | (_) { "no suffix" } }
middle = (s : String) { s |> ("[" <> rest <> "]") { rest } | (_) { "no pair" } }

main = () {
	stdio.print(core.inspect((prefix("pre:a"), suffix("main.i"), middle("[b]"))))
}
```

```text
("a", "main", "b")
```

One binder captures the variable text; fixed parts cannot overlap, and patterns cannot interpolate.

## Literal patterns

```i run package=tour title="src/main.i"

	i/stdio

literal = (x : { Int | Float | String }) {
	x |> (-1) { "integer" } | (-.1.5) { "float" } | ("ok") { "string" } | (_) { "other" }

main = () {
	stdio.print(core.inspect((literal(-1), literal(-.1.5), literal("ok"), literal(2))))
}
```

```text
("integer", "float", "string", "other")
```

## Bit patterns

```i run package=tour title="src/main.i"
(
	i/core
	i/stdio
)

packet = (bits : BitArray) {
	bits |> (<<tag, body:bits>>) { (tag, body) } | (_) { (0, <<>>) }
}

main = () {
	stdio.print(core.inspect(packet(<<7, 8, 9>>)))
	stdio.print(core.inspect(packet(<<>>)))
}
```

```text
(7, <<8, 9>>)
(0, <<>>)
```

A bit pattern consumes the whole input; an unmarked segment reads eight bits.

## Coverage

```i error E0516 package=tour title="src/main.i"
describe = (n : Int) { n |> (0) { "zero" } | (1) { "one" } }
main = () { describe(2) }
```

```text
E0516
```

Add an unguarded catch-all or narrow the input.

A lone unguarded branch instead reports a failed pattern through `Assert`; see [services](/docs/tour/services-and-errors).

## Recursion

```i run package=tour title="src/main.i"
(
	i/core
	i/stdio
)

sum = (xs : List(Int), total : Int) -> Int {
	xs |> ([]) { total } | ([x, ..rest]) { sum(rest, total + x) }
}

main = () {
	stdio.print(
		core.inspect(sum([1, 2, 3], 0)),
	)
}
```

```text
6
```

The recursive call is in tail position: no work follows it, so it uses no additional stack.

## Pipelines

```i run package=tour title="src/main.i"
(
	i/stdio
	std/int
)

Num = (Int)

Add = (Expr, Expr)
Mul = (Expr, Expr)
Expr = Num | Add | Mul
eval = (e) {
	e |>
		(Num(n)) { n }
		| (Add(a, b)) { eval(a) + eval(b) }
		| (Mul(a, b)) { eval(a) * eval(b) }
}

main = () {
	sum = Add(Num(1), Num(5))
	expr = Mul(sum, Num(7))
	expr |> eval |> int.to_string |> stdio.print
}
```

```text
42
```

`|>` passes a result to argument position zero of the next call.

## Grouping

```i run package=tour title="src/main.i"
(
	i/core
	i/stdio
)

main = () {
	stdio.print(core.inspect(2 + 3 * 4))
	stdio.print(core.inspect({ 2 + 3 } * 4))
}
```

```text
14
20
```

Equal-precedence operators associate left to right; calls and fields bind above arithmetic, pipelines below it.
