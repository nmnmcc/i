```i run package=hello title="src/main.i"
(
	i/stdio
	i/int
)

Ready = ()
Waiting = (Int)
message =
	(Ready) { "ready" }
	& (Waiting(minutes)) { int.to_string(minutes) <> " min left" }

main = () {
	stdio.print(message(Ready))
	stdio.print(message(Waiting(2)))
}
```

```text title="i run"
ready
2 min left
```

<section className="example-panel" id="panel-safe" aria-labelledby="title-safe">
```i types project=examples/deadline package=hello title="src/main.i"
(
	i/fail
	(Clock, now) = time/clock
)

(
	remaining
	Late(..)
)

Late = ()
remaining = (due) {
	due - now() |> (n ? n < 0) { fail.raise(Late) } & (n) { n }
}
```

```text title="i check --types"
hello/main.remaining : (Int) <- (Clock, Fail(Late)) -> Int
```

<section className="example-panel" id="panel-testable" aria-labelledby="title-testable">
```i test stdout=exact project=examples/deadline package=hello title="test/deadline_test.i"
(
	testlib = (test) = i/test
	i/fail
	app = hello/main
	(Clock) = time/clock
)

test("a fake clock", () {
	{
		assert(app.remaining(42) == 2)
		expired = fail.catch(() { app.remaining(30) })
		assert(expired == Error(app.Late))
	} <- (Clock(now = () { 40 }))
})
```

```text title="i test"
pass hello/deadline_test.a fake clock
1 passed, 0 failed
```
</section>
