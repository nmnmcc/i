Dep { Dep(v: Int) }

get = (d ! Dep) { d.v }

S(N) { S(f: () -> Int ! N) }

T(A, N) { T(g: (A) -> A ! N, m: (N) -> Nil) }

use(N) = (s ! S(N)) {
  S(f) = s
  f()
}

uset(A, N) = (x: A, t ! T(A, N)) -> A ! N | T(A, N) {
  T(g, _) = t
  g(x)
}

run(A, N) = (body: () -> A ! N) { body() }

fw = () { use() }

fr = () { run(() { use() }) }

ping = (n: Int) { n |> {
  0 -> use()
  _ -> pong(n - 1)
} }

pong = (n: Int) { ping(n) + uset(1) }

both(N) = () -> Int ! N | S(N) | T(Int, N) { pong(2) }

wrap(N) = (body: () -> Int ! N | S(N)) -> Int ! N {
  ! S(N) = S(() { 1 })
  body()
}

deep = () { wrap(() { fr() + get() }) }

closed = () {
  ! Dep = Dep(1)
  ! T(Int, Dep) = T((x: Int) { x + get() }, (_: Dep) { Nil })
  ! S(Dep) = S(() { get() })
  both() + deep()
}

bad(N) = (s ! S(N)) -> Int ! N {
  S(f) = s
  f()
}

loop = (x) { x(x) }

@main
main = () {
  closed()
  Nil
}
